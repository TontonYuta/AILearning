\chapter{Giải Phẫu Toàn Diện Kiến Trúc Transformer \& Cơ Chế Attention}
\label{ch:transformer-architecture}

Năm 2017, bài báo lịch sử mang tựa đề đầy khiêu khích \textit{``Attention Is All You Need''} của nhóm nghiên cứu Google Brain và Google Research (Vaswani et al.) đã khai sinh ra kiến trúc \textbf{Transformer}. Sự xuất hiện của nó đã quét sạch các kiến trúc mạng hồi quy (RNN, LSTM, GRU) vốn thống trị xử lý ngôn ngữ tự nhiên suốt nhiều thập kỷ trước đó, đặt nền móng duy nhất cho toàn bộ cuộc cách mạng Mô hình Ngôn ngữ Lớn (LLMs) ngày nay.

Đối với một Kỹ sư AI, việc hiểu Transformer không thể dừng lại ở việc đọc sơ đồ khối. Bạn phải thấu hiểu:
\begin{enumerate}
  \item Bản chất giải tích tại sao phép tính chú ý phải chia cho $\sqrt{d_k}$ để giữ ổn định dòng gradient.
  \item Nút thắt cổ chai bộ nhớ phần cứng GPU (Memory-bound) và cách đột phá \textbf{FlashAttention} sử dụng thuật toán \textbf{Online Softmax} để vượt qua giới hạn bộ nhớ HBM.
  \item Cơ chế mã hóa vị trí quay \textbf{RoPE} trong không gian phức và các kỹ thuật mở rộng ngữ cảnh dài (YaRN, NTK-aware).
  \item Sự tiến hóa tối ưu hóa bộ nhớ suy luận từ Multi-Head Attention (MHA) sang Grouped-Query Attention (GQA).
\end{enumerate}

\section{Sự Sụp Đổ Của Mạng Hồi Quy (RNN/LSTM) \& Động Cơ Của Attention}
\label{sec:rnn-failure-and-motivation}

Trước khi Transformer ra đời, mạng nơ-ron hồi quy (Recurrent Neural Networks -- RNN) và các biến thể có cổng nhớ như LSTM (Long Short-Term Memory) là lựa chọn tiêu chuẩn cho dữ liệu chuỗi. Tuy nhiên, kiến trúc RNN mang trong mình hai điểm nghẽn vật lý không thể khắc phục:

\paragraph{1. Điểm nghẽn Tuần tự Hóa (Sequential Computation Bottleneck)}:
Trong RNN, trạng thái ẩn $\mathbf{h}_t$ tại bước thời gian $t$ bắt buộc phải đợi trạng thái ẩn $\mathbf{h}_{t-1}$ của bước trước đó hoàn tất:
\begin{equation}
\mathbf{h}_t = \tanh(\mathbf{W}_{hh} \mathbf{h}_{t-1} + \mathbf{W}_{xh} \mathbf{x}_t + \mathbf{b})
\end{equation}
Quy trình này biến việc tính toán thành một chuỗi tuần tự nghiêm ngặt. Hệ quả là: \textbf{Không thể tận dụng năng lực tính toán song song của hàng chục nghìn nhân GPU}. Cho dù bạn có cụm 8 GPU H100 hiện đại nhất, RNN vẫn phải duyệt qua từng từ một, khiến thời gian huấn luyện trên các tập dữ liệu nghìn tỷ token trở thành bất khả thi.

\paragraph{2. Điểm nghẽn Cổ chai Thông tin (Information Bottleneck) \& Mất trí nhớ dài hạn}:
Trong kiến trúc Encoder-Decoder của RNN, toàn bộ ý nghĩa của một văn bản dài $1.000$ từ bị ép phải nén vào đúng một véc-tơ trạng thái ẩn cuối cùng $\mathbf{h}_T$ có kích thước cố định (ví dụ $512$ chiều). Khi chuỗi văn bản dài ra, thông tin từ các từ đầu tiên bị pha loãng dần theo cấp số nhân qua chuỗi các phép nhân ma trận.

\begin{definition}[Đột Phá Của Cơ Chế Tự Chú Ý -- Self-Attention]
Cơ chế Tự chú ý loại bỏ hoàn toàn khái niệm thời gian tuần tự. Thay vì truyền thông tin qua một chuỗi các bước ẩn trung gian, nó cho phép \textbf{mọi từ trong câu kết nối trực tiếp với mọi từ khác trong câu sau đúng một bước tính toán duy nhất}, với khoảng cách đường truyền thông tin giữa hai vị trí bất kỳ luôn bằng $O(1)$.
\end{definition}

\section{Bộ Ba Thần Thánh Q, K, V \& Giải Tích Toán Học Hệ Số $\frac{1}{\sqrt{d_k}}$}
\label{sec:qkv-and-attention-math}

\subsection{Ý Nghĩa Trực Giác Của Query, Key, Value}
Để hiểu tại sao phép tính Self-Attention lại sử dụng ba ma trận $\mathbf{Q}, \mathbf{K}, \mathbf{V}$, hãy liên tưởng tới hệ thống tra cứu video trên YouTube hoặc cơ sở dữ liệu:
\begin{itemize}
  \item \textbf{Query ($\mathbf{Q}$ -- Câu hỏi tìm kiếm)}: Đại diện cho thứ mà token hiện tại đang tìm kiếm để bổ sung ý nghĩa cho chính nó.
  \item \textbf{Key ($\mathbf{K}$ -- Nhãn từ khóa)}: Đại diện cho nội dung hoặc thuộc tính mà mỗi token khác sở hữu, đóng vai trò như nhãn chỉ mục để đối chiếu với Query.
  \item \textbf{Value ($\mathbf{V}$ -- Giá trị nội dung)}: Đại diện cho thông tin thực sự sẽ được trích xuất và tổng hợp nếu Query và Key khớp nhau.
\end{itemize}

Từ một véc-tơ đầu vào $\mathbf{x}_i \in \mathbb{R}^{d_{\text{model}}}$ của token thứ $i$, mô hình chiếu nó qua 3 ma trận trọng số học được:
\begin{equation}
\mathbf{q}_i = \mathbf{W}_Q \mathbf{x}_i, \qquad \mathbf{k}_i = \mathbf{W}_K \mathbf{x}_i, \qquad \mathbf{v}_i = \mathbf{W}_V \mathbf{x}_i
\end{equation}
Trong đó $\mathbf{W}_Q, \mathbf{W}_K \in \mathbb{R}^{d_k \times d_{\text{model}}}$ và $\mathbf{W}_V \in \mathbb{R}^{d_v \times d_{\text{model}}}$.

\subsection{Công Thức Toàn Diện Của Scaled Dot-Product Attention}
Dưới dạng ma trận cho toàn bộ chuỗi có độ dài $L$, gom các véc-tơ thành ma trận $\mathbf{Q} \in \mathbb{R}^{L \times d_k}$, $\mathbf{K} \in \mathbb{R}^{L \times d_k}$, và $\mathbf{V} \in \mathbb{R}^{L \times d_v}$:
\begin{equation}
\text{Attention}(\mathbf{Q}, \mathbf{K}, \mathbf{V}) = \text{Softmax}\left( \frac{\mathbf{Q} \mathbf{K}^T}{\sqrt{d_k}} \right) \mathbf{V}
\label{eq:scaled-dot-product-attention}
\end{equation}

\subsection{Chứng Minh Toán Học: Tại Sao Phải Chia Cho $\sqrt{d_k}$?}
Rất nhiều tài liệu chỉ ghi chép công thức mà không giải thích tại sao lại là $\sqrt{d_k}$ chứ không phải $d_k$ hay $\ln(d_k)$. Dưới đây là chứng minh giải tích chuẩn xác:

\begin{theorem}[Phương Sai Của Tích Vô Hướng Hai Véc-tơ Độc Lập]
Giả sử các phần tử $q_i$ và $k_i$ ($i = 1, \dots, d_k$) của véc-tơ Query và Key là các biến ngẫu nhiên độc lập, cùng phân phối với kỳ vọng $\mathbb{E}[q_i] = \mathbb{E}[k_i] = 0$ và phương sai $\text{Var}(q_i) = \text{Var}(k_i) = 1$.
Khi đó, tích vô hướng $S = \mathbf{q} \cdot \mathbf{k} = \sum_{i=1}^{d_k} q_i k_i$ có:
\begin{align}
\mathbb{E}[S] &= 0 \\
\text{Var}(S) &= d_k
\end{align}
\end{theorem}

\begin{proof}
Tính kỳ vọng của từng số hạng $z_i = q_i k_i$:
Vì $q_i$ và $k_i$ độc lập: $\mathbb{E}[z_i] = \mathbb{E}[q_i] \mathbb{E}[k_i] = 0 \times 0 = 0$.
Do đó:
\begin{equation}
\mathbb{E}[S] = \sum_{i=1}^{d_k} \mathbb{E}[z_i] = 0
\end{equation}
Tính phương sai của từng số hạng:
\begin{equation}
\text{Var}(z_i) = \mathbb{E}[z_i^2] - (\mathbb{E}[z_i])^2 = \mathbb{E}[q_i^2 k_i^2] - 0 = \mathbb{E}[q_i^2] \mathbb{E}[k_i^2]
\end{equation}
Vì $\text{Var}(q_i) = \mathbb{E}[q_i^2] - (\mathbb{E}[q_i])^2 = 1 \implies \mathbb{E}[q_i^2] = 1$. Tương tự $\mathbb{E}[k_i^2] = 1$.
Suy ra:
\begin{equation}
\text{Var}(z_i) = 1 \times 1 = 1
\end{equation}
Vì các $z_i$ độc lập với nhau, phương sai của tổng bằng tổng các phương sai:
\begin{equation}
\text{Var}(S) = \text{Var}\left(\sum_{i=1}^{d_k} z_i\right) = \sum_{i=1}^{d_k} \text{Var}(z_i) = \sum_{i=1}^{d_k} 1 = d_k
\end{equation}
Độ lệch chuẩn của tích vô hướng là $\sigma_S = \sqrt{\text{Var}(S)} = \sqrt{d_k}$.
\end{proof}

\paragraph{Hệ Lụy Của Hiện Tượng Bão Hòa Hàm Softmax Khi $d_k$ Lớn}:
Khi $d_k = 128$, độ lệch chuẩn của điểm số chú ý thô là $\sqrt{128} \approx 11.31$.
Các giá trị trong ma trận $\mathbf{S} = \mathbf{Q}\mathbf{K}^T$ sẽ dao động rất rộng (ví dụ từ $-30$ đến $+30$).
Khi đưa các giá trị có độ chênh lệch hàng chục đơn vị vào hàm Softmax:
\begin{equation*}
\text{Softmax}(\mathbf{z})_i = \frac{e^{z_i}}{\sum_j e^{z_j}}
\end{equation*}
Phần tử lớn nhất sẽ chiếm trọn xác suất $\approx 1.0$, trong khi tất cả các phần tử khác bị đẩy về gần $0$.
Đạo hàm của hàm Softmax đối với đầu vào là:
\begin{equation}
\frac{\partial \text{Softmax}_i}{\partial z_j} = \text{Softmax}_i (\delta_{ij} - \text{Softmax}_j)
\end{equation}
Khi $\text{Softmax}_i \approx 1$ và $\text{Softmax}_j \approx 0$, đạo hàm này triệt tiêu về $0$!
Toàn bộ gradient truyền ngược qua lớp Attention bị nghẽn lại (\textbf{Vanishing Gradient}), khiến mô hình hoàn toàn ngừng học!
Bằng cách chia cho $\sqrt{d_k}$, phương sai của điểm số được chuẩn hóa trở lại đúng bằng $1$:
\begin{equation}
\text{Var}\left(\frac{S}{\sqrt{d_k}}\right) = \frac{\text{Var}(S)}{(\sqrt{d_k})^2} = \frac{d_k}{d_k} = 1
\end{equation}
Giúp các giá trị rơi đúng vào vùng có đạo hàm nhạy cảm nhất của hàm Softmax!

\section{Đột Phá Phần Cứng: FlashAttention \& Thuật Toán Online Softmax}
\label{sec:flashattention-deepdive}

Để hiểu tại sao FlashAttention (Tri Dao et al., 2022) lại tạo nên một cuộc cách mạng trong kỹ nghệ AI, chúng ta phải nhìn vào kiến trúc vật lý của chip GPU:

Trong thuật toán Attention truyền thống:
\begin{enumerate}
  \item Đọc $\mathbf{Q}, \mathbf{K}$ từ HBM vào SRAM $\to$ Tính $\mathbf{S} = \mathbf{Q}\mathbf{K}^T \in \mathbb{R}^{L \times L}$.
  \item \textbf{Ghi ma trận $L \times L$ ra HBM}!
  \item Đọc lại ma trận $L \times L$ từ HBM vào SRAM $\to$ Tính Softmax $\mathbf{A} \in \mathbb{R}^{L \times L}$.
  \item \textbf{Ghi tiếp ma trận $L \times L$ ra HBM}!
  \item Đọc $\mathbf{A}$ và $\mathbf{V}$ từ HBM vào SRAM $\to$ Tính $\mathbf{O} = \mathbf{A}\mathbf{V} \in \mathbb{R}^{L \times d_v} \to$ Ghi $\mathbf{O}$ ra HBM.
\end{enumerate}
Khi $L = 32.000$, ma trận $L \times L$ chiếm hàng Gigabyte bộ nhớ. Thời gian GPU ngồi chờ đọc/ghi dữ liệu qua lại giữa HBM và SRAM chiếm tới $80\%$ tổng thời gian thực thi, trong khi các nhân tính toán Tensor Cores bị bỏ đói! Phép tính Attention bị giới hạn bởi bộ nhớ (\textbf{Memory-Bound}) chứ không phải năng lực tính toán.

\subsection{Thuật Toán Online Softmax Của FlashAttention}
Rào cản lớn nhất ngăn cản việc tính toán Attention theo từng khối nhỏ (Tiling) trực tiếp trên SRAM là hàm Softmax: Mẫu số của Softmax $\sum_{j=1}^L e^{z_j}$ đòi hỏi phải nhìn thấy toàn bộ hàng trước khi có thể chia tỷ lệ!

FlashAttention sử dụng thuật toán \term{Online Softmax} (Milakov \& Gimelshein, 2018) cho phép cập nhật giá trị Softmax tích lũy theo từng khối mà không cần lưu trữ toàn bộ ma trận:

\begin{definition}[Quy Tắc Cập Nhật Online Softmax Theo Từng Khối]
Cho một véc-tơ đầu vào được chia thành hai khối: $\mathbf{x} = [\mathbf{x}^{(1)}, \mathbf{x}^{(2)}]$.
Giả sử ta đã tính được giá trị cực đại cục bộ $m^{(1)} = \max(\mathbf{x}^{(1)})$ và tổng lũy thừa chuẩn hóa $d^{(1)} = \sum_j e^{x_j^{(1)} - m^{(1)}}$.
Khi nạp thêm khối thứ hai $\mathbf{x}^{(2)}$ với cực đại $m^{(2)} = \max(\mathbf{x}^{(2)})$ và tổng $d^{(2)} = \sum_j e^{x_j^{(2)} - m^{(2)}}$:
\begin{enumerate}
  \item Cập nhật giá trị cực đại toàn cục mới:
  \begin{equation}
  m^{\text{new}} = \max\left(m^{(1)}, m^{(2)}\right)
  \end{equation}
  \item Cập nhật tổng mẫu số mới bằng cách hiệu chỉnh tỷ lệ tổng cũ:
  \begin{equation}
  d^{\text{new}} = d^{(1)} \cdot e^{m^{(1)} - m^{\text{new}}} + d^{(2)} \cdot e^{m^{(2)} - m^{\text{new}}}
  \end{equation}
  \item Véc-tơ đầu ra tích lũy $\mathbf{o}^{\text{new}}$ được cập nhật theo hệ số hiệu chỉnh:
  \begin{equation}
  \mathbf{o}^{\text{new}} = \mathbf{o}^{(1)} \cdot \left(\frac{d^{(1)} e^{m^{(1)} - m^{\text{new}}}}{d^{\text{new}}}\right) + \mathbf{o}^{(2)} \cdot \left(\frac{e^{m^{(2)} - m^{\text{new}}}}{d^{\text{new}}}\right)
  \end{equation}
\end{enumerate}
\end{definition}

Nhờ thuật toán này, GPU nạp từng khối nhỏ của $\mathbf{Q}, \mathbf{K}, \mathbf{V}$ vào SRAM, tính toán xong tích lũy ngay vào kết quả đầu ra $\mathbf{O}$, \textbf{hoàn toàn không bao giờ ghi ma trận $L \times L$ ra HBM}! FlashAttention giảm số lần đọc ghi bộ nhớ từ $O(L^2)$ xuống $O(L)$, tăng tốc độ từ $2\times$ đến $4\times$ lần và cho phép chạy ngữ cảnh dài gấp hàng chục lần trên cùng một phần cứng!

\section{Mã Hóa Vị Trí Hiện Đại: Từ Sinusoidal Đến RoPE (Rotary Position Embedding)}
\label{sec:rope-deepdive}

Vì phép tính Self-Attention hoàn toàn là phép nhân ma trận trên các tập hợp, nó có tính chất \textbf{Hoán vị Bất biến (Permutation Invariant)}: Nếu bạn xáo trộn thứ tự các từ trong câu, kết quả tính toán của Attention không thay đổi! Để mô hình nhận biết từ nào đứng trước, từ nào đứng sau, chúng ta bắt buộc phải tiêm thông tin vị trí vào véc-tơ biểu diễn.

\subsection{Hạn Chế Của Sinusoidal Positional Encoding Cổ Điển}
Trong Transformer 2017, Vaswani et al. sử dụng các hàm sóng $\sin$ và $\cos$ với các tần số khác nhau:
\begin{equation}
PE_{(pos, 2i)} = \sin\left(\frac{pos}{10000^{2i/d}}\right), \qquad PE_{(pos, 2i+1)} = \cos\left(\frac{pos}{10000^{2i/d}}\right)
\end{equation}
Véc-tơ vị trí này được cộng trực tiếp vào véc-tơ nhúng từ vựng: $\mathbf{x}_{\text{final}} = \mathbf{x}_{\text{embed}} + \mathbf{PE}$.
Điểm yếu: Phép cộng này làm biến dạng không gian biểu diễn ngữ nghĩa của véc-tơ gốc và không khái quát tốt khi gặp độ dài ngữ cảnh vượt quá độ dài huấn luyện ban đầu.

\subsection{Bản Chất Toán Học Của RoPE (Rotary Position Embedding)}
Được đề xuất bởi Su et al. (2021) và được áp dụng trong $100\%$ các LLM hàng đầu hiện nay (Llama, Mistral, Qwen, Gemma), \textbf{RoPE} mã hóa vị trí bằng \textbf{phép quay hình học trong không gian phức}.

Thay vì cộng véc-tơ, RoPE xoay véc-tơ Query và Key một góc tỷ lệ thuận với vị trí của nó. Chia không gian $d$-chiều thành $\frac{d}{2}$ cặp tọa độ 2D độc lập. Tại vị trí $m$, mỗi cặp tọa độ $(x_1, x_2)$ được nhân với ma trận quay 2D:
\begin{equation}
\mathbf{R}_{\theta_i, m} = \begin{bmatrix} \cos(m\theta_i) & -\sin(m\theta_i) \\ \sin(m\theta_i) & \cos(m\theta_i) \end{bmatrix}, \quad \text{với } \theta_i = 10000^{-2(i-1)/d}
\label{eq:rope-2d-matrix}
\end{equation}

\begin{theorem}[Bản Chất Tương Đối Tuyệt Đối Của RoPE]
Tích vô hướng giữa véc-tơ Query tại vị trí $m$ và véc-tơ Key tại vị trí $n$ sau khi áp dụng RoPE chỉ phụ thuộc vào \textbf{khoảng cách tương đối} $(m - n)$ giữa hai từ, hoàn toàn độc lập với vị trí tuyệt đối của chúng:
\begin{equation}
\langle \mathbf{R}_m \mathbf{q}, \mathbf{R}_n \mathbf{k} \rangle = (\mathbf{R}_m \mathbf{q})^T (\mathbf{R}_n \mathbf{k}) = \mathbf{q}^T \mathbf{R}_m^T \mathbf{R}_n \mathbf{k} = \mathbf{q}^T \mathbf{R}_{n-m} \mathbf{k} = g(\mathbf{q}, \mathbf{k}, m - n)
\label{eq:rope-relative-distance-proof}
\end{equation}
\end{theorem}

\subsection{Kỹ Thuật Mở Rộng Ngữ Cảnh Dài (Long-Context RoPE Scaling)}
Khi muốn mở rộng một mô hình được huấn luyện ở ngữ cảnh $4\text{K}$ lên $32\text{K}$ hoặc $128\text{K}$ tokens mà không cần huấn luyện lại từ đầu:
\begin{itemize}
  \item \textbf{Linear RoPE Scaling}: Chia vị trí $m$ cho hệ số mở rộng $s = \frac{L_{\text{target}}}{L_{\text{train}}}$: $m' = \frac{m}{s}$. Phương pháp này nén toàn bộ phân phối góc nhưng làm mất độ phân giải ở các từ đứng sát nhau.
  \item \textbf{NTK-Aware Scaled RoPE}: Dựa trên lý thuyết Neural Tangent Kernel, nhận thấy rằng các chiều có tần số cao mã hóa thông tin vị trí cục bộ cực kỳ chính xác (không nên nén), trong khi các chiều tần số thấp mã hóa thông tin khoảng cách xa (cần nén). NTK-aware điều chỉnh tần số cơ sở $\theta_{\text{base}}$ từ $10.000$ lên $500.000$ hoặc $1.000.000$, giữ nguyên độ phân giải cục bộ mà vẫn mở rộng ngữ cảnh mượt mà!
  \item \textbf{YaRN (Yet another RoPE extensioN)}: Kết hợp nội suy NTK với việc điều chỉnh nhiệt độ phân phối Attention Entropy, cho phép mở rộng ngữ cảnh lên tới $128\text{K}$ tokens chỉ sau vài trăm bước fine-tuning!
\end{itemize}

\section{Các Biến Thể Attention Hiện Đại: MHA vs MQA vs GQA}
\label{sec:mha-mqa-gqa}

Trong giai đoạn suy luận tự hồi quy (Decode), dung lượng bộ nhớ VRAM bị chiếm dụng bởi \textbf{KV Cache} là rào cản lớn nhất ngăn cản việc phục vụ nhiều người dùng đồng thời. Sự tiến hóa của các kiến trúc Attention chủ yếu nhằm mục đích nén bộ nhớ KV Cache này:

\label{fig:mha-mqa-gqa-comparison}
\end{figure}

\begin{itemize}
  \item \textbf{Multi-Head Attention (MHA)}: Mỗi Head Query sở hữu một Head Key và một Head Value riêng biệt ($H_Q = H_K = H_V = 32$). Chiếm dụng VRAM khổng lồ.
  \item \textbf{Multi-Query Attention (MQA)}: Tất cả các Head Query chia sẻ chung đúng $1$ Head Key và $1$ Head Value ($H_Q = 32, H_K = H_V = 1$). Tiết kiệm 32 lần bộ nhớ nhưng làm suy giảm năng lực lập luận phức tạp.
  \item \textbf{Grouped-Query Attention (GQA)}: Dung hòa hoàn hảo: Chia các Head Query thành $G$ nhóm, mỗi nhóm chia sẻ một cặp KV Head. Ví dụ trong Llama-3-70B: $H_Q = 64$ Query heads chia thành 8 nhóm, mỗi nhóm dùng chung 1 cặp KV head ($H_{KV} = 8$). Bộ nhớ KV Cache giảm đúng \textbf{8 lần} so với MHA mà độ chính xác gần như tương đương $100\%$!
\end{itemize}

\section{Ví Dụ Giải Mẫu Tính Toán Số Học Bằng Tay}
\label{sec:ch06-numerical-walkthrough}

\begin{example}[Tính Toán Toàn Bộ Ma Trận Self-Attention Bằng Tay Cho Chuỗi 3 Tokens]
Xét cơ chế Self-Attention đơn giản với $d_k = 2$ cho chuỗi gồm 3 tokens ($L=3$):
\begin{equation*}
\mathbf{Q} = \begin{bmatrix} 1.0 & 0.0 \\ 0.0 & 2.0 \\ 1.0 & 1.0 \end{bmatrix}, \quad
\mathbf{K} = \begin{bmatrix} 1.0 & 1.0 \\ 2.0 & 0.0 \\ 0.0 & 1.0 \end{bmatrix}, \quad
\mathbf{V} = \begin{bmatrix} 10.0 & 0.0 \\ 0.0 & 20.0 \\ 5.0 & 5.0 \end{bmatrix} \in \mathbb{R}^{3 \times 2}
\end{equation*}

\paragraph{Bước 1: Tính tích ma trận $\mathbf{S} = \mathbf{Q} \mathbf{K}^T$}
\begin{equation*}
\mathbf{S} = \begin{bmatrix} 1.0 & 0.0 \\ 0.0 & 2.0 \\ 1.0 & 1.0 \end{bmatrix} \begin{bmatrix} 1.0 & 2.0 & 0.0 \\ 1.0 & 0.0 & 1.0 \end{bmatrix} = \begin{bmatrix} 1.0 & 2.0 & 0.0 \\ 2.0 & 0.0 & 2.0 \\ 2.0 & 2.0 & 1.0 \end{bmatrix}
\end{equation*}

\paragraph{Bước 2: Chuẩn hóa theo tỷ lệ $\sqrt{d_k} = \sqrt{2} \approx 1.4142$}
\begin{equation*}
\mathbf{S}_{\text{scaled}} = \frac{\mathbf{S}}{\sqrt{2}} \approx \begin{bmatrix} 0.7071 & 1.4142 & 0.0000 \\ 1.4142 & 0.0000 & 1.4142 \\ 1.4142 & 1.4142 & 0.7071 \end{bmatrix}
\end{equation*}

\paragraph{Bước 3: Tính ma trận trọng số Softmax $\mathbf{A}$}
Xét hàng thứ nhất: $\mathbf{s}_1 = [0.7071, 1.4142, 0.0000]$.
Trừ giá trị lớn nhất $c = 1.4142$:
\begin{align*}
e^{0.7071 - 1.4142} &= e^{-0.7071} \approx 0.4931 \\
e^{1.4142 - 1.4142} &= e^{0.0} = 1.0000 \\
e^{0.0000 - 1.4142} &= e^{-1.4142} \approx 0.2431 \\
\text{Tổng hàng 1} &= 0.4931 + 1.0000 + 0.2431 = 1.7362
\end{align*}
Trọng số chú ý hàng 1:
\begin{equation*}
A_{11} = \frac{0.4931}{1.7362} \approx 0.2840, \quad A_{12} = \frac{1.0000}{1.7362} \approx 0.5760, \quad A_{13} = \frac{0.2431}{1.7362} \approx 0.1400
\end{equation*}

\paragraph{Bước 4: Tính véc-tơ đầu ra $\mathbf{o}_1$}
\begin{align*}
\mathbf{o}_1 &= 0.2840 \begin{bmatrix} 10.0 \\ 0.0 \end{bmatrix} + 0.5760 \begin{bmatrix} 0.0 \\ 20.0 \end{bmatrix} + 0.1400 \begin{bmatrix} 5.0 \\ 5.0 \end{bmatrix} \\
&= \begin{bmatrix} 2.840 \\ 0.000 \end{bmatrix} + \begin{bmatrix} 0.000 \\ 11.520 \end{bmatrix} + \begin{bmatrix} 0.700 \\ 0.700 \end{bmatrix} = \begin{bmatrix} 3.540 \\ 12.220 \end{bmatrix}
\end{align*}
\end{example}

\begin{example}[Thuật Toán Online Softmax Từng Bước]
Xét véc-tơ gồm 4 phần tử $\mathbf{x} = [2.0, 4.0, 1.0, 5.0]$, chia làm 2 block với kích thước $B = 2$:
\begin{itemize}
  \item Block 1: $\mathbf{x}^{(1)} = [2.0, 4.0]$.
  \item Block 2: $\mathbf{x}^{(2)} = [1.0, 5.0]$.
\end{itemize}
\paragraph{Xử lý Block 1}:
\begin{itemize}
  \item $m_1 = \max(2.0, 4.0) = 4.0$.
  \item $d_1 = e^{2.0 - 4.0} + e^{4.0 - 4.0} = e^{-2.0} + e^{0.0} \approx 0.1353 + 1.0 = 1.1353$.
\end{itemize}
\paragraph{Xử lý Block 2}:
\begin{itemize}
  \item $m_{\text{block2}} = \max(1.0, 5.0) = 5.0$.
  \item $m_2 = \max(m_1, m_{\text{block2}}) = \max(4.0, 5.0) = 5.0$.
  \item Cập nhật mẫu số:
  \begin{align*}
  d_2 &= d_1 \cdot e^{m_1 - m_2} + \left( e^{1.0 - 5.0} + e^{5.0 - 5.0} \right) \\
  &= 1.1353 \cdot e^{4.0 - 5.0} + \left( e^{-4.0} + e^{0.0} \right) \\
  &= 1.1353 \cdot 0.3679 + (0.0183 + 1.0) \approx 0.4177 + 1.0183 = 1.4360
  \end{align*}
\end{itemize}
\paragraph{Tính xác suất Softmax cuối cùng}:
\begin{align*}
S_1 &= \frac{e^{2.0 - 5.0}}{d_2} = \frac{e^{-3.0}}{1.4360} = \frac{0.0498}{1.4360} \approx \mathbf{0.0347} \\
S_2 &= \frac{e^{4.0 - 5.0}}{d_2} = \frac{e^{-1.0}}{1.4360} = \frac{0.3679}{1.4360} \approx \mathbf{0.2562} \\
S_3 &= \frac{e^{1.0 - 5.0}}{d_2} = \frac{e^{-4.0}}{1.4360} = \frac{0.0183}{1.4360} \approx \mathbf{0.0127} \\
S_4 &= \frac{e^{5.0 - 5.0}}{d_2} = \frac{e^{0.0}}{1.4360} = \frac{1.0000}{1.4360} \approx \mathbf{0.6964}
\end{align*}
Tổng xác suất: $0.0347 + 0.2562 + 0.0127 + 0.6964 = 1.0000$. Kết quả trùng khớp hoàn hảo với Softmax chuẩn mà không cần lưu toàn bộ mảng cùng lúc!
\end{example}

\section{Hiện Thực Hóa: Thuật Toán Attention, Online Softmax \& RoPE}
\label{sec:ch06-implementation}

Dưới đây là mã nguồn Python chuẩn mực triển khai thuật toán Scaled Dot-Product Attention có Causal Masking, thuật toán Online Softmax (FlashAttention primitive) và phép quay RoPE 2D:

\begin{lstlisting}[language=Python]
import numpy as np
from typing import Tuple

class ScaledDotProductAttention:
    """Cai dat Attention kem Causal Masking bang NumPy."""
    @staticmethod
    def forward(
        Q: np.ndarray,
        K: np.ndarray,
        V: np.ndarray,
        is_causal: bool = False
    ) -> Tuple[np.ndarray, np.ndarray]:
        d_k = Q.shape[-1]
        scores = np.matmul(Q, np.swapaxes(K, -1, -2)) / np.sqrt(d_k)
        
        if is_causal:
            seq_len_q = Q.shape[-2]
            seq_len_k = K.shape[-2]
            mask = np.triu(np.ones((seq_len_q, seq_len_k)), k=1).astype(bool)
            scores = np.where(mask, -1e4, scores)  # -1e4 an toan cho FP16
            
        shifted_scores = scores - np.max(scores, axis=-1, keepdims=True)
        exp_scores = np.exp(shifted_scores)
        attention_weights = exp_scores / np.sum(exp_scores, axis=-1, keepdims=True)
        
        output = np.matmul(attention_weights, V)
        return output, attention_weights

def online_softmax(x: np.ndarray, block_size: int = 2) -> np.ndarray:
    """Tinh Softmax truc tuyen theo tung block (FlashAttention Primitive)."""
    n = len(x)
    m = -np.inf
    d = 0.0
    
    for i in range(0, n, block_size):
        chunk = x[i:i + block_size]
        chunk_max = np.max(chunk)
        m_new = max(m, chunk_max)
        d = d * np.exp(m - m_new) + np.sum(np.exp(chunk - m_new))
        m = m_new

    return np.exp(x - m) / d

def apply_rotary_2d(vec: np.ndarray, pos: int, base: float = 10000.0) -> np.ndarray:
    """Ap dung phep quay vi tri RoPE 2D len tung cap toa do."""
    dim = len(vec)
    rotated = np.zeros_like(vec)
    for i in range(0, dim, 2):
        theta = 1.0 / (base ** (i / dim))
        angle = pos * theta
        cos_val = np.cos(angle)
        sin_val = np.sin(angle)
        
        x0, x1 = vec[i], vec[i + 1]
        rotated[i] = x0 * cos_val - x1 * sin_val
        rotated[i + 1] = x0 * sin_val + x1 * cos_val
    return rotated
\end{lstlisting}

\section{Hướng Dẫn Lời Giải \& Bình Luận Kỹ Thuật Chi Tiết}
\label{sec:ch06-solutions}

\subsection{Lời Giải Nhóm A: Câu Hỏi Tư Duy Bản Chất}
\begin{itemize}
  \item \textbf{Lời giải Câu A.1}:
  Trong giai đoạn huấn luyện, mục tiêu của mô hình tự hồi quy là dự đoán token tiếp theo $x_{t+1}$ dựa trên các token quá khứ $x_1, \dots, x_t$. Nhờ kiến trúc Transformer, toàn bộ $L$ tokens được đưa vào song song cùng một lúc.
  Nếu không có Causal Masking, tại vị trí $t$, token $x_t$ sẽ nhìn thấy được véc-tơ Key và Value của token tương lai $x_{t+1}$ ngay trong ma trận chú ý $\mathbf{Q} \mathbf{K}^T$. Mô hình sẽ ``gian lận'' bằng cách sao chép trực tiếp token tương lai sang đầu ra mà không cần học bất kỳ quy luật ngữ nghĩa nào!
  Khi đưa vào môi trường suy luận thực tế (Inference), token tương lai chưa hề tồn tại, mô hình sẽ hoàn toàn tê liệt và không thể sinh được văn bản có nghĩa. Giá trị $-\infty$ ép xác suất Softmax tại các vị trí tương lai về đúng $e^{-\infty} = 0$, ngăn chặn tuyệt đối việc nhìn thấy trước tương lai.

  \item \textbf{Lời giải Câu A.2}:
  Trong Post-LN: $\mathbf{x}_{l+1} = \text{LayerNorm}(\mathbf{x}_l + \text{SubLayer}(\mathbf{x}_l))$. Đường truyền tắt (Residual Connection) bị chặn bởi lớp LayerNorm. Khi lan truyền ngược gradient qua nhiều tầng, gradient bị co giãn liên tục qua đạo hàm của LayerNorm, dẫn tới mất ổn định ở các tầng đầu (cần learning rate warmup rất dài).
  Trong Pre-LN: $\mathbf{x}_{l+1} = \mathbf{x}_l + \text{SubLayer}(\text{LayerNorm}(\mathbf{x}_l))$. Đường truyền tắt là một đường cao tốc thuần túy nối thẳng từ tầng đầu tới tầng cuối: $\mathbf{x}_L = \mathbf{x}_0 + \sum_{l=0}^{L-1} \text{SubLayer}(\text{LayerNorm}(\mathbf{x}_l))$. Gradient có thể chảy thẳng về $\mathbf{x}_0$ với hệ số 1 mà không bị cản trở, giúp quá trình huấn luyện cực kỳ ổn định ngay cả với các mô hình hàng trăm tầng.

  \item \textbf{Lời giải Câu A.3}:
  - \textbf{ALiBi}: Trừ độ lệch tuyến tính $m(i - j)$ vào ma trận attention. Ngoại suy độ dài rất tốt mà không cần huấn luyện lại, nhưng không nắm bắt tốt các quan hệ tuần hoàn ngữ nghĩa và không linh hoạt bằng RoPE trong việc điều chỉnh phân giải theo từng dải tần số.
  - \textbf{RoPE}: Mã hóa vị trí bằng phép quay hình học. Khi kết hợp với các kỹ thuật nội suy tần số (NTK-Aware, YaRN), RoPE cho phép mở rộng ngữ cảnh lên tới hàng trăm nghìn token với độ chính xác vượt trội.

  \item \textbf{Lời giải Câu A.4}:
  Standard Attention bị giới hạn bởi tốc độ đọc/ghi bộ nhớ HBM (Memory-Bound). Chip GPU dành phần lớn thời gian để vận chuyển ma trận $L \times L$ khổng lồ qua lại giữa HBM và SRAM. FlashAttention sử dụng nguyên lý Tiling kết hợp Online Softmax để tính toán toàn bộ Attention trực tiếp trên bộ nhớ đệm SRAM siêu nhanh mà không bao giờ ghi ma trận trung gian ra HBM. Tốc độ tăng vọt là nhờ loại bỏ hoàn toàn chi phí I/O đọc/ghi bộ nhớ chậm, chứ không phải do giảm số phép tính FLOPs.

  \item \textbf{Lời giải Câu A.5}:
  Linear RoPE chia đều tất cả các tần số góc cho hệ số $s$. Điều này làm giảm bước sóng của các tần số cao (vốn chịu trách nhiệm phân biệt vị trí giữa các từ liền kề), khiến mô hình bị ``mờ mắt'' ở cự ly gần. NTK-Aware giữ nguyên các tần số cao (bảo toàn độ phân giải khoảng cách ngắn) và chỉ kéo giãn các tần số thấp (mở rộng khả năng bao quát khoảng cách dài).
\end{itemize}

\subsection{Lời Giải Nhóm B: Bài Toán Tính Toán Giải Tích Bằng Tay}

\paragraph{Lời giải Bài B.1}:
Công thức tính dung lượng KV Cache cho toàn bộ hệ thống:
\begin{equation*}
\text{Memory}_{\text{KV}} = 2 \times (\text{Layers}) \times (H_{KV}) \times (d_k) \times (\text{Context Length}) \times (\text{Batch}) \times (\text{Bytes/element})
\end{equation*}
\begin{itemize}
  \item[a)] Với Multi-Head Attention ($H_{KV} = 32$):
  \begin{align*}
  \text{Memory}_{\text{KV, MHA}} &= 2 \times 32 \times 32 \times 128 \times 8.192 \times 4 \times 2\text{ bytes} \\
  &= 17.179.869.184\text{ bytes} = \frac{17.179.869.184}{1024^3}\text{ GiB} = \mathbf{16.0\text{ GiB}}
  \end{align*}
  \item[b)] Với Grouped-Query Attention ($H_{KV} = 8$):
  \begin{equation*}
  \text{Memory}_{\text{KV, GQA}} = \frac{16.0\text{ GiB}}{4} = \mathbf{4.0\text{ GiB}}
  \end{equation*}
  Dung lượng VRAM tiết kiệm được là: $16.0 - 4.0 = \mathbf{12.0\text{ GiB}}$!
\end{itemize}

\paragraph{Lời giải Bài B.2}:
Đặt $S = \mathbf{u} \cdot \mathbf{v} = \sum_{i=1}^d u_i v_i$.
Vì $u_i, v_i$ độc lập và có kỳ vọng 0: $\mathbb{E}[u_i v_i] = \mathbb{E}[u_i]\mathbb{E}[v_i] = 0 \implies \mathbb{E}[S] = 0$.
Phương sai của từng tích $z_i = u_i v_i$:
\begin{equation*}
\text{Var}(z_i) = \mathbb{E}[u_i^2 v_i^2] - (\mathbb{E}[u_i v_i])^2 = \mathbb{E}[u_i^2]\mathbb{E}[v_i^2] - 0 = \sigma^2 \times \sigma^2 = \sigma^4
\end{equation*}
Vì các $z_i$ độc lập: $\text{Var}(S) = \sum_{i=1}^d \text{Var}(z_i) = \sum_{i=1}^d \sigma^4 = d \cdot \sigma^4$. (Điều phải chứng minh).

\paragraph{Lời giải Bài B.3}:
Với $\mathbf{z} = [3.0, 1.0, 4.0, 2.0]$ và $B = 2$:
\begin{itemize}
  \item Block 1: $[3.0, 1.0] \implies m_1 = 3.0$, $d_1 = e^{3.0-3.0} + e^{1.0-3.0} = 1 + e^{-2} \approx 1 + 0.1353 = 1.1353$.
  \item Block 2: $[4.0, 2.0] \implies m_{\text{block2}} = 4.0$.
  \item Cập nhật toàn cục: $m_2 = \max(3.0, 4.0) = 4.0$.
  \item $d_2 = d_1 e^{3.0-4.0} + (e^{4.0-4.0} + e^{2.0-4.0}) = 1.1353 e^{-1} + (1 + e^{-2}) \approx 1.1353(0.3679) + 1.1353 \approx 0.4177 + 1.1353 = 1.5530$.
  \item Xác suất Softmax:
  $P_1 = \frac{e^{3-4}}{1.553} = \frac{0.3679}{1.553} \approx 0.2369$;
  $P_2 = \frac{e^{1-4}}{1.553} = \frac{0.0498}{1.553} \approx 0.0321$;
  $P_3 = \frac{e^{4-4}}{1.553} = \frac{1.0}{1.553} \approx 0.6439$;
  $P_4 = \frac{e^{2-4}}{1.553} = \frac{0.1353}{1.553} \approx 0.0871$.
\end{itemize}

\subsection{Lời Giải Nhóm C: Thực Hành Kỹ Nghệ \& Lập Trình}

\subsection{Lời Giải Nhóm D: Phân Tích Thất Bại Thực Tế (Post-Mortem Analysis)}

\begin{debugbox}[Báo Cáo Sự Cố 1: Tràn Số Âm FP16 Do Đặt Sai Giá Trị Mask]
\textbf{1. Tóm tắt sự cố}: Mô hình sinh ra toàn bộ ký tự \code{????????} do ma trận Softmax chứa giá trị \code{NaN}.
\textbf{2. Phân tích nguyên nhân gốc rễ}:
Chuẩn IEEE 754 float16 chỉ biểu diễn số âm nhỏ nhất tới $\approx -65.504$. Khi đặt \code{mask = -1e30}, giá trị bị tràn thành \code{-inf}. Khi thực hiện phép trừ giá trị cực đại để ổn định số học, phép toán $(-\infty) - (-\infty) = \text{NaN}$ phát sinh và lan truyền toàn mạng.
\textbf{3. Giải pháp khắc phục}:
Đặt giá trị mask an toàn cho FP16 là \code{-1e4} (tức $-10.000$), vì $e^{-10.000} = 0$ hoàn hảo và nằm an toàn trong dải của FP16.
\end{debugbox}

\begin{debugbox}[Báo Cáo Sự Cố 2: Suy Giảm Độ Chính Xác Cự Ly Gần Khi Dùng Linear RoPE]
\textbf{1. Tóm tắt sự cố}: Mô hình mở rộng ngữ cảnh 4K lên 32K bằng Linear RoPE bị mất trí nhớ ở các câu hỏi cự ly gần ($< 1\text{K}$ tokens).
\textbf{2. Phân tích nguyên nhân gốc rễ}:
Linear RoPE chia đều mọi tần số cho $s = 8$. Điều này làm các thành phần tần số cao bị nén quá mức, khiến hai từ đứng cạnh nhau (ví dụ từ ở vị trí 10 và 11) có góc quay chênh lệch quá nhỏ, mô hình không còn phân biệt được thứ tự ngữ pháp cục bộ.
\textbf{3. Giải pháp khắc phục}:
Chuyển sang kỹ thuật \textbf{NTK-Aware Scaling} hoặc \textbf{YaRN}: Giữ nguyên các tần số cao (bảo vệ thông tin ngữ pháp cự ly ngắn) và chỉ nén các tần số thấp (kéo dài khoảng cách bao quát toàn cục).
\end{debugbox}
