\chapter{Vận Hành Suy Luận LLM, Tokenomics \& Tối Ưu KV Cache}
\label{ch:llm-runtime-tokenomics}

Huấn luyện một Mô hình Ngôn ngữ Lớn (LLM) và phục vụ nó trong môi trường sản xuất (Production Serving) là hai bài toán kỹ thuật thuộc hai thế giới hoàn toàn khác biệt. Trong khi giai đoạn tiền huấn luyện (Pre-training) tập trung vào việc tối đa hóa thông lượng tính toán thô (FLOPS Utilization) trên các cụm siêu máy tính hàng nghìn GPU, thì giai đoạn suy luận (Inference Serving) lại bị chi phối bởi các ràng buộc khắt khe về độ trễ thời gian thực (Latency SLA) và chi phí tài chính hạ tầng (FinOps).

Một Kỹ sư AI xuất sắc không chỉ biết gọi API của nhà cung cấp đóng gói sẵn. Bạn phải thấu hiểu:
\begin{enumerate}
  \item Bản chất vật lý của hai pha vận hành: \textbf{Prefill (Compute-bound)} và \textbf{Decode (Memory-bound)}.
  \item Giải phẫu từng byte bộ nhớ của \textbf{KV Cache} và cách thuật toán \textbf{PagedAttention} triệt tiêu phân mảnh bộ nhớ từ $80\%$ xuống dưới $4\%$.
  \item Cơ chế tăng tốc suy luận phỏng đoán \textbf{Speculative Decoding} và kỹ thuật chia nhỏ \textbf{Chunked Prefill} giải quyết hiện tượng nghẽn đầu hàng (Head-of-Line Blocking).
  \item Các chiến lược lấy mẫu giải mã xác suất (Greedy, Temperature, Top-K, Top-P, Min-P) và bài toán kinh tế học token (Tokenomics ROI).
\end{enumerate}

\section{Giải Phẫu Quá Trình Suy Luận Tự Hồi Quy (Autoregressive Generation)}
\label{sec:autoregressive-anatomy}

Hầu hết các LLM tiên tiến hiện nay (như GPT-4, Llama-3, Mistral, Qwen) đều thuộc họ kiến trúc \textbf{Decoder-only Transformer}. Bản chất toán học của chúng là mô hình hóa phân phối xác suất có điều kiện của token tiếp theo dựa trên toàn bộ chuỗi token đã xuất hiện trước đó:
\begin{equation}
P(w_1, w_2, \dots, w_T) = \prod_{t=1}^T P(w_t \mid w_1, w_2, \dots, w_{t-1})
\end{equation}

Để sinh ra câu trả lời cho người dùng, hệ thống suy luận LLM bắt buộc phải trải qua hai giai đoạn vận hành có đặc tính phần cứng hoàn toàn trái ngược nhau:

\subsection{Pha 1: Đánh Giá Câu Lệnh Đầu Vào (Prefill Phase / Prompt Evaluation)}
Khi người dùng gửi một đoạn văn bản đầu vào (Prompt) gồm $L_{\text{prompt}}$ tokens:
\begin{itemize}
  \item \textbf{Cơ chế tính toán}: Mô hình xử lý toàn bộ $L_{\text{prompt}}$ tokens này \textbf{song song trong một lượt tính toán duy nhất} (Single Forward Pass). Nhờ cơ chế Causal Masking, tất cả các phép chiếu Query, Key, Value và phép nhân ma trận Attention được thực hiện đồng thời trên toàn bộ ma trận chuỗi.
  \item \textbf{Đặc tính phần cứng}: Đây là giai đoạn \textbf{Compute-bound} (Nghẽn tại năng lực tính toán). Tỷ số cường độ số học (Arithmetic Intensity) rất cao vì các nhân ma trận lớn cho phép nhân Tensor Cores của GPU hoạt động ở mức hiệu suất đỉnh cao.
  \item \textbf{Chỉ số đo lường}: Thời gian hoàn thành pha này quyết định trực tiếp chỉ số \textbf{TTFT (Time To First Token)} -- khoảng thời gian người dùng phải chờ đợi kể từ lúc bấm nút gửi đến khi ký tự đầu tiên xuất hiện trên màn hình.
\end{itemize}

\subsection{Pha 2: Sinh Token Tự Hồi Quy (Decode Phase / Token Generation)}
Sau khi đã sinh ra token đầu tiên, mô hình bước vào vòng lặp sinh tuần tự:
\begin{itemize}
  \item \textbf{Cơ chế tính toán}: Mỗi bước thời gian $t$, mô hình chỉ nhận đúng \textbf{1 token mới nhất} làm đầu vào để dự đoán token tiếp theo $t+1$. Quá trình này lặp đi lặp lại một cách tuần tự cho đến khi mô hình phát ra token kết thúc (End-of-Sequence -- \code{EOS}) hoặc chạm giới hạn độ dài tối đa.
  \item \textbf{Đặc tính phần cứng}: Đây là giai đoạn \textbf{Memory-bound} (Nghẽn tại băng thông bộ nhớ). Tại mỗi bước sinh một token duy nhất, toàn bộ hàng chục tỷ tham số của mô hình (Weights) bắt buộc phải được đọc tuần tự từ bộ nhớ VRAM HBM (High Bandwidth Memory) nạp vào bộ nhớ đệm SRAM của chip GPU chỉ để thực hiện một vài phép nhân ma trận với véc-tơ đơn lẻ.
  \item \textbf{Cường độ số học cực thấp}: Với mỗi tham số 16-bit (2 bytes), chip GPU chỉ thực hiện đúng 2 phép tính FLOPs (một phép nhân và một phép cộng). Cường độ số học của pha Decode rơi xuống mức thảm hại:
  \begin{equation}
  \text{Arithmetic Intensity}_{\text{Decode}} = \frac{2\text{ FLOPs}}{2\text{ Bytes}} = 1\text{ FLOP/Byte}
  \end{equation}
  Trong khi đó, GPU Nvidia H100 SXM có năng lực tính toán FP16 lên tới $989\text{ TFLOPS}$ nhưng băng thông bộ nhớ chỉ đạt $3.350\text{ GB/s}$. Điểm cân bằng của H100 đòi hỏi cường độ số học phải đạt ít nhất:
  \begin{equation}
  \frac{989 \times 10^{12}\text{ FLOPs/s}}{3.35 \times 10^{12}\text{ Bytes/s}} \approx 295\text{ FLOPs/Byte}
  \end{equation}
  Vì $1 \ll 295$, trong suốt pha Decode, hơn $99\%$ năng lực tính toán của các nhân Tensor Cores trên GPU bị bỏ đói, hoàn toàn nằm chờ dữ liệu được kéo qua bus bộ nhớ!
  \item \textbf{Chỉ số đo lường}: Tốc độ của pha này quyết định chỉ số \textbf{TPOT (Time Per Output Token)} -- độ trễ giữa các token liên tiếp, phản ánh tốc độ sinh chữ (Tokens/giây) mà người dùng cảm nhận được.
\end{itemize}

\section{Giải Phẫu Bộ Nhớ KV Cache \& Thách Thức Phân Mảnh}
\label{sec:kv-cache-anatomy}

\subsection{Tại Sao KV Cache Là Cứu Tinh Duy Nhất Của Pha Decode?}
Để tính toán token thứ $t$, mô hình cần nhân véc-tơ Query $\mathbf{q}_t$ với ma trận Key $\mathbf{K}_{1:t}$ và nhân tiếp với ma trận Value $\mathbf{V}_{1:t}$:
\begin{equation}
\mathbf{o}_t = \text{Softmax}\left(\frac{\mathbf{q}_t \mathbf{K}_{1:t}^T}{\sqrt{d_k}}\right) \mathbf{V}_{1:t}
\end{equation}

Nếu không lưu trữ lại dữ liệu, tại mỗi bước sinh token mới, mô hình sẽ phải tính toán lại véc-tơ Key $\mathbf{k}_i$ và Value $\mathbf{v}_i$ cho toàn bộ các token quá khứ từ $1$ đến $t-1$.
Để sinh một chuỗi dài $L$ tokens, tổng số phép tính Attention sẽ là:
\begin{equation}
\sum_{t=1}^L O(t) = O(L^2) \text{ phép tính trên mỗi token} \implies \text{Tổng chi phí: } O(L^3)
\end{equation}
Độ phức tạp bậc ba $O(L^3)$ sẽ biến việc sinh một đoạn văn bản vài nghìn token thành thảm họa độ trễ kéo dài hàng giờ đồng hồ!

\begin{definition}[KV Cache]
\textbf{KV Cache} là kỹ thuật lưu trữ cố định các véc-tơ Key ($\mathbf{k}$) và Value ($\mathbf{v}$) của tất cả các token đã xử lý trong các tầng Transformer vào bộ nhớ VRAM của GPU. Tại mỗi bước Decode tiếp theo, mô hình chỉ cần tính toán đúng một cặp $\mathbf{k}_t, \mathbf{v}_t$ cho token hiện tại, nối thêm (Append) vào bộ nhớ đệm KV Cache sẵn có, và thực hiện phép nhân véc-tơ -- ma trận.
\end{definition}

Nhờ KV Cache, chi phí tính toán tại mỗi bước decode giảm triệt để từ $O(t)$ xuống còn phép tính ma trận $O(1)$, đưa tổng độ phức tạp của toàn bộ quá trình sinh văn bản về mức tuyến tính $O(L)$.

\subsection{Công Thức Giải Tích Tính Dung Lượng KV Cache}
\begin{theorem}[Dung lượng bộ nhớ KV Cache]
Xét một mô hình Transformer có $n_{\text{layers}}$ tầng, cơ chế Attention có $n_{\text{kv\_heads}}$ đầu Key-Value (với MHA thì $n_{\text{kv\_heads}} = n_{\text{heads}}$, với GQA thì $n_{\text{kv\_heads}} < n_{\text{heads}}$), mỗi đầu có chiều dài $d_{\text{head}}$, và dữ liệu được lưu ở định dạng số thực có kích thước $b_{\text{bytes}}$ bytes (FP16/BF16: 2 bytes, FP8: 1 byte).
Dung lượng bộ nhớ VRAM cần thiết để lưu trữ KV Cache cho đúng \textbf{1 token} là:
\begin{equation}
M_{\text{token}} = 2 \times n_{\text{layers}} \times n_{\text{kv\_heads}} \times d_{\text{head}} \times b_{\text{bytes}} \quad (\text{Bytes/token})
\label{eq:kv-cache-per-token}
\end{equation}
(Hệ số 2 là do phải lưu cả hai ma trận Key và Value).
Với một lô phục vụ gồm $B$ yêu cầu đồng thời, mỗi yêu cầu có độ dài ngữ cảnh trung bình là $L$ tokens, tổng dung lượng VRAM bị KV Cache chiếm dụng là:
\begin{equation}
M_{\text{total\_KV}} = B \times L \times M_{\text{token}}
\label{eq:kv-cache-total}
\end{equation}
\end{theorem}

\section{Cuộc Cách Mạng PagedAttention \& Động Cơ vLLM}
\label{sec:pagedattention-vllm}

Trước khi vLLM ra đời (Kwon et al., UC Berkeley, 2023), các hệ thống suy luận LLM quản lý bộ nhớ KV Cache theo cách truyền thống: Cấp phát một mảng nhớ liên tục (Contiguous Memory Allocation) có kích thước bằng độ dài tối đa của mô hình (ví dụ: cấp phát trước cho $2.048$ hoặc $8.192$ tokens) cho mỗi request người dùng.

Cách làm ngây thơ này gây ra thảm họa lãng phí bộ nhớ:
\begin{enumerate}
  \item \textbf{Phân mảnh nội bộ (Internal Fragmentation)}: Cấp phát trước cho $2.048$ tokens nhưng người dùng chỉ hỏi một câu ngắn và nhận câu trả lời $100$ tokens $\implies$ Lãng phí hơn $95\%$ bộ nhớ đã đặt chỗ trước!
  \item \textbf{Phân mảnh bên ngoài (External Fragmentation)}: Khi các request có độ dài khác nhau hoàn thành và giải phóng bộ nhớ ở các thời điểm ngẫu nhiên, VRAM bị chia cắt thành hàng trăm lỗ hổng bộ nhớ nhỏ rời rạc không thể ghép lại.
\end{enumerate}
Nghiên cứu của nhóm UC Berkeley chỉ ra rằng: Trong các hệ thống LLM phục vụ truyền thống, \textbf{60\% đến 80\% dung lượng VRAM bị lãng phí do phân mảnh bộ nhớ}!

\subsection{Nguyên Lý Hoạt Động Của PagedAttention}
Lấy cảm hứng từ cơ chế \textbf{Phân trang Bộ nhớ Ảo (Virtual Memory Paging)} trong hệ điều hành kinh điển, \textbf{PagedAttention} giải quyết triệt để bài toán này:

\end{itemize}

\section{Suy Luận Phỏng Đoán (Speculative Decoding) \& Chunked Prefill}
\label{sec:speculative-decoding-chunked-prefill}

\subsection{Speculative Decoding: Phá Vỡ Bức Tường Memory-Bound}
Như đã phân tích, pha Decode bị chậm vì GPU tốn thời gian nạp trọng số mô hình lớn (Target Model, ví dụ Llama-3-70B) chỉ để sinh từng token đơn lẻ.
Để giải quyết bài toán này, Leviathan et al. (2023) đã phát minh ra kỹ thuật \term{Speculative Decoding} (Suy luận phỏng đoán):

\centering

    draft/.style={rectangle, draw=secondarycolor, thick, fill=accentcolor!20, align=center, rounded corners=2pt, minimum width=9.0cm, minimum height=0.8cm, font=\small},

\end{tikzpicture}
\caption{Quy trình suy luận phỏng đoán kết hợp Draft Model và Target Model}
\label{fig:speculative-decoding-flow}
\end{figure}

\begin{enumerate}
  \item \textbf{Sinh phỏng đoán (Draft Generation)}: Một mô hình phụ trợ nhỏ và nhanh (Draft Model, ví dụ Llama-3-1B) sinh nhanh $K$ token phỏng đoán tuần tự: $x_1, \dots, x_K$.
  \item \textbf{Kiểm chứng song song (Parallel Verification)}: Đưa toàn bộ $K$ token này vào mô hình lớn (Target Model, ví dụ Llama-3-70B) trong \textbf{đúng một lượt forward pass duy nhất}. Vì đây là phép tính ma trận trên $K$ token đồng thời, Target Model hoạt động ở chế độ Compute-bound hiệu suất cao!
  \item \textbf{Lấy mẫu chấp nhận (Modified Rejection Sampling)}:
  Với mỗi token $x_i$ có xác suất sinh bởi Draft Model là $q(x_i)$ và Target Model là $p(x_i)$:
  \begin{itemize}
    \item Nếu $p(x_i) \ge q(x_i)$: Chấp nhận token $x_i$ vô điều kiện.
    \item Nếu $p(x_i) < q(x_i)$: Chấp nhận với xác suất $\frac{p(x_i)}{q(x_i)}$. Nếu bị từ chối, hủy bỏ các token phỏng đoán phía sau và lấy mẫu lại từ phân phối hiệu chỉnh $\max(0, p(x) - q(x))$.
  \end{itemize}
\end{enumerate}

\begin{theorem}[Bảo Toàn Phân Phối Của Speculative Decoding]
Thuật toán lấy mẫu loại bỏ hiệu chỉnh của Leviathan et al. bảo đảm toán học rằng: Phân phối xác suất của chuỗi token sinh ra bởi Speculative Decoding \textbf{đồng nhất 100\%} với phân phối xác suất gốc của Target Model:
\begin{equation}
P_{\text{speculative}}(w) \equiv P_{\text{target}}(w)
\end{equation}
Nghĩa là việc dùng mô hình Draft nhỏ chỉ có tác dụng tăng tốc độ ($2\times$ đến $3\times$ lần) mà tuyệt đối không làm suy giảm chất lượng đầu ra của mô hình lớn!
\end{theorem}

\subsection{Chứng Minh Giải Tích: Tính Đồng Nhất Phân Phối Của Speculative Decoding}
\label{sec:speculative-math-proof}

Để hiểu tại sao việc dùng một mô hình nhỏ 1B phỏng đoán mà không làm giảm trí thông minh của mô hình lớn 70B, chúng ta hãy thực hiện chứng minh giải tích chặt chẽ theo phân tích của Leviathan et al. (2023) và Chen et al. (2023).

\begin{proof}
Xét một vị trí sinh token bất kỳ trong không gian từ vựng rời rạc $\mathcal{V}$.
Gọi $q(x)$ là phân phối xác suất sinh ra bởi Draft Model $M_q$ và $p(x)$ là phân phối xác suất sinh ra bởi Target Model $M_p$.
Theo quy trình lấy mẫu loại bỏ hiệu chỉnh (Modified Rejection Sampling):
\begin{enumerate}
  \item Mô hình Draft sinh ra một token ứng viên $x \sim q(x)$.
  \item Token $x$ được Target Model chấp nhận với xác suất:
  \begin{equation}
  \alpha(x) = \min\left(1, \frac{p(x)}{q(x)}\right)
  \end{equation}
  \item Nếu token $x$ bị từ chối (xảy ra khi $p(x) < q(x)$), hệ thống hủy bỏ $x$ và lấy mẫu lại một token mới từ phân phối phần dư chuẩn hóa:
  \begin{equation}
  p'(x) = \frac{\max(0, p(x) - q(x))}{\sum_{y \in \mathcal{V}} \max(0, p(y) - q(y))}
  \end{equation}
\end{enumerate}

Bây giờ, hãy tính xác suất toàn phần để token $x$ được sinh ra ở đầu ra cuối cùng, ký hiệu là $P(X = x)$:
\begin{equation}
P(X = x) = P(\text{chấp nhận } x) + P(\text{bị từ chối}) \times P(\text{sinh ra } x \text{ trong pha lấy mẫu lại})
\end{equation}

\textbf{Bước 1: Tính xác suất chấp nhận $x$ trực tiếp}:
Xác suất để Draft Model chọn $x$ và Target Model chấp nhận $x$ là:
\begin{equation}
P(\text{chấp nhận } x) = q(x) \times \min\left(1, \frac{p(x)}{q(x)}\right) = \min(q(x), p(x))
\end{equation}

\textbf{Bước 2: Tính tổng xác suất bị từ chối toàn cục}:
Tổng xác suất chấp nhận trên toàn bộ từ vựng là:
\begin{equation}
\beta = \sum_{y \in \mathcal{V}} P(\text{chấp nhận } y) = \sum_{y \in \mathcal{V}} \min(q(y), p(y))
\end{equation}
Do đó, xác suất xảy ra biến cố từ chối là:
\begin{equation}
P(\text{bị từ chối}) = 1 - \beta = 1 - \sum_{y \in \mathcal{V}} \min(q(y), p(y))
\end{equation}
Mặt khác, với mọi số thực $a, b$: $\max(0, a - b) = a - \min(a, b)$. Vì $\sum_y p(y) = 1$, ta có:
\begin{equation}
\sum_{y \in \mathcal{V}} \max(0, p(y) - q(y)) = \sum_{y \in \mathcal{V}} \left[ p(y) - \min(q(y), p(y)) \right] = 1 - \sum_{y \in \mathcal{V}} \min(q(y), p(y)) = 1 - \beta
\end{equation}
\textbf{Mẫu số chuẩn hóa của $p'(x)$ bằng chính xác xác suất bị từ chối $1 - \beta$!}

\textbf{Bước 3: Hợp nhất xác suất toàn phần}:
Thay các đại lượng vào công thức $P(X = x)$:
\begin{align}
P(X = x) &= \min(q(x), p(x)) + (1 - \beta) \times \frac{\max(0, p(x) - q(x))}{1 - \beta} \nonumber \\
&= \min(q(x), p(x)) + \max(0, p(x) - q(x))
\end{align}
Vì với mọi cặp số thực $a, b$, ta luôn có đẳng thức cơ bản: $\min(a, b) + \max(0, a - b) = a$.
Áp dụng cho $a = p(x)$ và $b = q(x)$:
\begin{equation}
P(X = x) = p(x), \qquad \forall x \in \mathcal{V}
\end{equation}
Phân phối đầu ra của Speculative Decoding hoàn toàn trùng khớp $100\%$ với phân phối gốc của Target Model!
\end{proof}

\paragraph{Mối liên hệ giữa Tỷ lệ chấp nhận và Khoảng cách Biến thiên Toàn phần (Total Variation Distance)}:
Tỷ lệ chấp nhận trung bình $\alpha$ của một token phỏng đoán được liên kết trực tiếp với khoảng cách phân phối:
\begin{equation}
\alpha = \sum_{x \in \mathcal{V}} \min(q(x), p(x)) = 1 - \frac{1}{2} \sum_{x \in \mathcal{V}} |p(x) - q(x)| = 1 - \text{TVD}(p, q)
\end{equation}
Khi mô hình Draft có năng lực mô phỏng Target Model càng cao, $\text{TVD} \to 0 \implies \alpha \to 1.0$, tốc độ sinh chữ tiến tới giới hạn lý thuyết tối đa $(K+1)$ lần!

\paragraph{Biến thể Medusa: Phỏng đoán đa đầu không cần Draft Model (Cai et al., 2024)}:
Một rào cản thực tế của Speculative Decoding truyền thống là hệ thống phải nạp cùng lúc hai mô hình độc lập vào VRAM (ví dụ Llama-70B và Llama-1B). Kiến trúc \textbf{Medusa} giải quyết bài toán này bằng cách đóng băng mô hình Target và chỉ huấn luyện thêm nhiều đầu chiếu tuyến tính (Medusa Heads) trên đỉnh của tầng Transformer cuối cùng:
\begin{itemize}
  \item Đầu Medusa thứ $k$ dự đoán token tại bước tương lai $t+k$.
  \item Không cần thêm Draft Model riêng biệt, tiết kiệm tối đa VRAM.
  \item Sử dụng đồ thị cây phỏng đoán (Tree Attention) để kiểm chứng đồng thời nhiều nhánh token ứng viên trong một lượt forward pass.
\end{itemize}

\subsection{RadixAttention \& Cấu Trúc Cây Tiền Tố (Radix Tree) Trong SGLang}
\label{sec:radixattention-sglang}

Mặc dù PagedAttention (vLLM) giải quyết hoàn hảo bài toán phân mảnh bộ nhớ bên trong một request, nó có một hạn chế lớn: Khi request kết thúc, toàn bộ các block KV Cache được giải phóng vào Free Pool. Trong môi trường doanh nghiệp thực tế, các yêu cầu tiếp theo thường xuyên chia sẻ những đoạn tiền tố khổng lồ:
\begin{itemize}
  \item Cuộc hội thoại nhiều lượt (Multi-turn Chatbot): Lượt chat thứ 5 lặp lại toàn bộ 4 lượt chat trước đó.
  \item Kỹ thuật RAG doanh nghiệp: Hàng trăm người dùng cùng truy vấn một tài liệu chính sách hoặc cẩm nang 50 trang.
  \item Tác nhân AI (AI Coding Agents): Các vòng lặp ReAct lặp lại toàn bộ System Prompt và cấu trúc cây thư mục dự án.
\end{itemize}

Được nhóm nghiên cứu UC Berkeley đề xuất trong công trình \textbf{SGLang} (Zheng et al., 2024), \textbf{RadixAttention} tổ chức toàn bộ bộ nhớ KV Cache trên GPU thành một cấu trúc cây tiền tố thu gọn (\textbf{Radix Tree}):
\begin{enumerate}
  \item \textbf{Khớp tiền tố dài nhất (Longest Prefix Match)}: Khi một request mới đến, hệ thống duyệt từ nút gốc của cây để tìm chuỗi token trùng lặp dài nhất. Thay vì tính toán lại, hệ thống tái sử dụng ngay $100\%$ các block KV Cache đã tính sẵn trong VRAM!
  \item \textbf{Phân tách nút linh hoạt (Node Splitting)}: Khi hai request có chung một đoạn đầu nhưng rẽ nhánh ở giữa, nút hiện tại được tự động chia đôi thành một nút cha chung và hai nút con độc lập.
  \item \textbf{Cơ chế thu hồi bộ nhớ theo LRU (Least Recently Used Eviction)}: Khi bộ nhớ VRAM chạm ngưỡng đầy, hệ thống không giải phóng ngẫu nhiên mà thực hiện tỉa (Prune) các nút lá ít được truy cập nhất, bảo toàn vĩnh viễn các nút gốc (như System Prompt hoặc tài liệu chung) trên GPU.
\end{enumerate}

\paragraph{Hiệu quả định lượng vượt bậc}:
Với các tác vụ Agent hoặc Chatbot nhiều lượt có tỷ lệ khớp tiền tố (Prefix Cache Hit Rate) đạt $80\% - 90\%$:
\begin{equation}
\text{TTFT}_{\text{cached}} \approx \frac{L_{\text{unmatched}}}{L_{\text{prompt}}} \times \text{TTFT}_{\text{cold}}
\end{equation}
Thời gian chờ ký tự đầu tiên (TTFT) giảm ngoạn mục từ $2.5\text{ giây}$ xuống chỉ còn dưới \textbf{$80\text{ mili-giây}$}, đồng thời cắt giảm tới $80\%$ lượng điện năng và năng lực tính toán FLOPs của cụm máy chủ!

\subsection{Chunked Prefill: Chống Nghẽn Đầu Hàng (Head-of-Line Blocking)}
Trong hệ thống phục vụ nhiều người dùng đồng thời, một vấn đề nhức nhối là hiện tượng \term{Head-of-Line Blocking}:
Khi hệ thống đang decode mượt mà cho 50 người dùng với độ trễ 20ms/token, bất ngờ có một request mới xuất hiện với prompt dài $16.000$ tokens. Pha Prefill của request mới này cần tới 500ms tính toán, chiếm trọn GPU và đóng băng việc sinh chữ của 50 người dùng cũ!

Giải pháp \textbf{Chunked Prefill (vLLM / Sarathi-Serve)} chia prompt dài thành các chunk cố định (ví dụ $512$ tokens). Tại mỗi bước lặp của GPU, hệ thống ghép chung các chunk prefill với các token decode của các request cũ, giữ cho độ trễ giữa các token (ITL) luôn duy trì ở mức ổn định.

\subsection{1. Bộ Lấy Mẫu Giải Mã Đa Năng (Decoding Sampler)}
\begin{lstlisting}[language=Python]
import math
import numpy as np
from typing import List, Tuple, Dict

class DecodingSampler:
    """Bo giai ma token ho tro Temperature, Top-K, Top-P, Min-P va Greedy."""
    @staticmethod
    def sample(
        logits: np.ndarray,
        temperature: float = 1.0,
        top_k: int = 0,
        top_p: float = 1.0,
        min_p: float = 0.0,
        rng: np.random.Generator = None
    ) -> int:
        if rng is None:
            rng = np.random.default_rng(42)

        logits = np.array(logits, dtype=np.float64)

        if temperature <= 1e-5:
            return int(np.argmax(logits))

        # 1. Temperature scaling & Stable Softmax
        scaled_logits = logits / temperature
        max_logit = np.max(scaled_logits)
        exp_logits = np.exp(scaled_logits - max_logit)
        probs = exp_logits / np.sum(exp_logits)

        # 2. Min-P Filtering
        if min_p > 0.0:
            threshold = min_p * np.max(probs)
            probs[probs < threshold] = 0.0

        # 3. Top-K Truncation
        if 0 < top_k < len(probs):
            top_k_indices = np.argpartition(probs, -top_k)[-top_k:]
            mask = np.ones(len(probs), dtype=bool)
            mask[top_k_indices] = False
            probs[mask] = 0.0

        # 4. Top-P (Nucleus) Truncation
        if top_p < 1.0:
            sorted_indices = np.argsort(probs)[::-1]
            sorted_probs = probs[sorted_indices]
            cumulative_probs = np.cumsum(sorted_probs)
            cutoff = np.searchsorted(cumulative_probs, top_p)
            cutoff = min(cutoff + 1, len(probs))
            valid = sorted_indices[:cutoff]
            mask = np.ones(len(probs), dtype=bool)
            mask[valid] = False
            probs[mask] = 0.0

        total_p = np.sum(probs)
        if total_p == 0.0:
            return int(np.argmax(logits))

        probs = probs / total_p
        return int(rng.choice(len(probs), p=probs))
\end{lstlisting}

class SpeculativeDecoderVerifier:
    """Mo phong Rejection Sampling cua Speculative Decoding (Leviathan et al.)."""
    @staticmethod
    def verify_tokens(
        draft_probs: List[float],
        target_probs: List[float],
        random_draws: List[float]
    ) -> Tuple[int, List[int]]:
        accepted = []
        for i, (q, p, u) in enumerate(zip(draft_probs, target_probs, random_draws)):
            if p >= q:
                accepted.append(i)
            elif u < (p / q):
                accepted.append(i)
            else:
                break
        return len(accepted), accepted

class ChunkedPrefillScheduler:
    """Lap lich xen ke giua Prefill chunks va Decode tokens de on dinh do tre ITL."""
    def __init__(self, max_tokens_per_iteration: int = 512):
        self.max_tokens_budget = max_tokens_per_iteration

    def schedule_iteration(
        self,
        active_decode_reqs: List[int],
        pending_prefill_reqs: List[Tuple[int, int]]  # (req_id, remaining_tokens)
    ) -> Tuple[List[int], List[Tuple[int, int]]]:
        # Uu tien phuc vu toan bo cac session dang Decode (1 token moi session)
        decode_budget = len(active_decode_reqs)
        decode_served = list(active_decode_reqs)
        
        remaining_budget = max(0, self.max_tokens_budget - decode_budget)
        prefill_served = []

        # Phan bo phan ngan sach con lai cho cac chunk prefill
        for req_id, remaining_tokens in pending_prefill_reqs:
            if remaining_budget <= 0:
                break
            chunk_size = min(remaining_tokens, remaining_budget)
            prefill_served.append((req_id, chunk_size))
            remaining_budget -= chunk_size

        return decode_served, prefill_served
\end{lstlisting}

\subsection{4. Cây Tiền Tố Radix Tree Prefix Cache (SGLang)}
\begin{lstlisting}[language=Python]
class RadixNode:
    """Nut trong Cay Tien to Radix Tree luu tru chuoi token va Block IDs."""
    def __init__(self, token_subsequence: List[int], block_ids: List[int], parent: Optional['RadixNode'] = None):
        self.token_subsequence = list(token_subsequence)
        self.block_ids = list(block_ids)
        self.parent = parent
        self.children: Dict[int, 'RadixNode'] = {}
        self.last_accessed: float = 0.0

    @property
    def is_leaf(self) -> bool:
        return len(self.children) == 0

class RadixCacheManager:
    """
    Quan ly bo nho dem tien to Radix Tree theo kien truc SGLang.
    Ho tro khop tien to dai nhat, phan tach nut (Node Splitting) va thu hoi LRU.
    """
    def __init__(self, max_blocks: int, block_size: int = 16):
        self.max_blocks = max_blocks
        self.block_size = block_size
        self.root = RadixNode(token_subsequence=[], block_ids=[])
        self.next_block_id = 0
        self.time_counter = 0.0

    def get_used_blocks(self) -> int:
        def count_blocks(node: RadixNode) -> int:
            return len(node.block_ids) + sum(count_blocks(child) for child in node.children.values())
        return count_blocks(self.root)

    def match_prefix(self, tokens: List[int]) -> Tuple[int, List[int]]:
        """Duyet cay de tim tien to trung khop dai nhat cho chuoi tokens."""
        self.time_counter += 1.0
        current = self.root
        matched_tokens = 0
        cached_blocks = []
        idx = 0

        while idx < len(tokens):
            first_tok = tokens[idx]
            if first_tok not in current.children:
                break
            child = current.children[first_tok]
            subseq = child.token_subsequence
            match_len = 0
            while match_len < len(subseq) and idx + match_len < len(tokens):
                if subseq[match_len] != tokens[idx + match_len]:
                    break
                match_len += 1

            if match_len == len(subseq):
                matched_tokens += match_len
                cached_blocks.extend(child.block_ids)
                child.last_accessed = self.time_counter
                current = child
                idx += match_len
            else:
                matched_tokens += match_len
                num_full_blocks = match_len // self.block_size
                cached_blocks.extend(child.block_ids[:num_full_blocks])
                child.last_accessed = self.time_counter
                break

        return matched_tokens, cached_blocks

    def insert(self, tokens: List[int]) -> List[int]:
        """Nap chuoi token moi vao cay Radix, tu dong chia tach nut neu re nhanh."""
        self.time_counter += 1.0
        current = self.root
        all_blocks = []
        idx = 0

        while idx < len(tokens):
            first_tok = tokens[idx]
            if first_tok not in current.children:
                rem_tokens = tokens[idx:]
                num_blocks = math.ceil(len(rem_tokens) / self.block_size)
                self._ensure_capacity(num_blocks)
                new_blocks = [self._alloc_block() for _ in range(num_blocks)]
                new_node = RadixNode(rem_tokens, new_blocks, parent=current)
                new_node.last_accessed = self.time_counter
                current.children[first_tok] = new_node
                all_blocks.extend(new_blocks)
                return all_blocks

            child = current.children[first_tok]
            subseq = child.token_subsequence
            match_len = 0
            while match_len < len(subseq) and idx + match_len < len(tokens):
                if subseq[match_len] != tokens[idx + match_len]:
                    break
                match_len += 1

            if match_len == len(subseq):
                all_blocks.extend(child.block_ids)
                child.last_accessed = self.time_counter
                current = child
                idx += match_len
            else:
                common_tokens = subseq[:match_len]
                rem_child_tokens = subseq[match_len:]
                rem_insert_tokens = tokens[idx + match_len:]

                num_common_blocks = match_len // self.block_size
                common_blocks = child.block_ids[:num_common_blocks]
                rem_child_blocks = child.block_ids[num_common_blocks:]

                split_node = RadixNode(common_tokens, common_blocks, parent=current)
                split_node.last_accessed = self.time_counter
                current.children[first_tok] = split_node

                child.token_subsequence = rem_child_tokens
                child.block_ids = rem_child_blocks
                child.parent = split_node
                split_node.children[rem_child_tokens[0]] = child

                all_blocks.extend(common_blocks)

                if rem_insert_tokens:
                    num_new_blocks = math.ceil(len(rem_insert_tokens) / self.block_size)
                    self._ensure_capacity(num_new_blocks)
                    new_blocks = [self._alloc_block() for _ in range(num_new_blocks)]
                    new_child = RadixNode(rem_insert_tokens, new_blocks, parent=split_node)
                    new_child.last_accessed = self.time_counter
                    split_node.children[rem_insert_tokens[0]] = new_child
                    all_blocks.extend(new_blocks)

                return all_blocks

        return all_blocks

    def _alloc_block(self) -> int:
        bid = self.next_block_id
        self.next_block_id += 1
        return bid

    def _ensure_capacity(self, needed: int):
        while self.get_used_blocks() + needed > self.max_blocks:
            leaves = []
            def collect_leaves(node: RadixNode):
                if node != self.root and node.is_leaf:
                    leaves.append(node)
                for c in node.children.values():
                    collect_leaves(c)
            collect_leaves(self.root)
            if not leaves:
                break
            lru_leaf = min(leaves, key=lambda n: n.last_accessed)
            del lru_leaf.parent.children[lru_leaf.token_subsequence[0]]
\end{lstlisting}

  Xét mô hình \textbf{Llama-3-70B} ($n_{\text{layers}} = 80$, $n_{\text{kv\_heads}} = 8$, $d_{\text{head}} = 128$).
  Doanh nghiệp muốn triển khai cụm phục vụ với cửa sổ ngữ cảnh mở rộng tối đa $L = 128.000$ tokens ở định dạng FP8 ($b = 1$ byte).

  \item \textbf{Bài B.2 (Chứng minh Giải tích Giới hạn Nhiệt độ)}:

  \item \textbf{Bài B.3 (Kỳ Vọng Số Lượng Token Được Chấp Nhận Trong Speculative Decoding)}:
  Xét một hệ thống Speculative Decoding sinh $K = 4$ token phỏng đoán tại mỗi lượt. Giả sử xác suất chấp nhận độc lập của mỗi token phỏng đoán là $\alpha = 0.8$.
  Hệ thống dừng chấp nhận ngay khi có một token bị từ chối và sinh thêm một token hiệu chỉnh từ Target Model.
  Tính số lượng token trung bình được sinh ra bởi Target Model sau mỗi bước lặp $\mathbb{E}[N]$.

  \item \textbf{Bài B.4 (Chứng Minh Giải Tích Tính Đồng Nhất Phân Phối Của Speculative Decoding)}:
  Cho từ vựng rời rạc $\mathcal{V}$, mô hình Draft có phân phối xác suất $q(x)$ và mô hình Target có phân phối $p(x)$ thỏa mãn $q(x) > 0, p(x) \ge 0, \sum_{x \in \mathcal{V}} q(x) = \sum_{x \in \mathcal{V}} p(x) = 1$.
  Xác suất chấp nhận ứng viên $x \sim q$ là $\alpha(x) = \min\left(1, \frac{p(x)}{q(x)}\right)$. Khi bị từ chối, token mới được lấy mẫu từ phân phối phần dư hiệu chỉnh:
  \begin{equation*}
  p'(x) = \frac{\max(0, p(x) - q(x))}{\sum_{y \in \mathcal{V}} \max(0, p(y) - q(y))}
  \end{equation*}
  \begin{itemize}
    \item[a)] Chứng minh xác suất chấp nhận kỳ vọng tại pha Draft là $\beta = \sum_{x \in \mathcal{V}} \min(p(x), q(x))$.
    \item[b)] Chứng minh mẫu số của $p'(x)$ thỏa mãn $\sum_{y \in \mathcal{V}} \max(0, p(y) - q(y)) = 1 - \beta$.
    \item[c)] Sử dụng định lý xác suất toàn phần để chứng minh $P(X = x) \equiv p(x)$ với mọi $x \in \mathcal{V}$.
  \end{itemize}

  \item \textbf{Bài B.5 (Phân Tích Định Lượng TTFT \& Tiết Kiệm FLOPs Của Radix Tree)}:
  Một hệ thống tiếp nhận $R = 100\text{ requests/giây}$. Mỗi request có độ dài prompt $L = 2.000$ tokens, trong đó phần tiền tố chung (System Prompt và Few-shot Examples) chiếm $L_{\text{prefix}} = 1.700$ tokens ($85\%$).
  Mô hình 70B FP8 tiêu tốn xấp xỉ $C = 140\text{ GFLOPs/token}$ trong pha Prefill. GPU H100 đạt thông lượng tính toán thực tế $1.000\text{ TFLOPs/s}$.
  \begin{itemize}
    \item[a)] Khi không có Cache tiền tố, tính tổng công suất tính toán (TFLOPs/s) cần thiết để phục vụ lượng request trên và thời gian tính toán Prefill lý thuyết cho mỗi request trên 1 GPU.
    \item[b)] Khi Radix Tree đạt tỷ lệ khớp $85\%$ cho toàn bộ các request, tính số TFLOPs được giải phóng mỗi giây và mức cải thiện thời gian Prefill.
  \end{itemize}

\section{Hướng Dẫn Lời Giải \& Bình Luận Kỹ Thuật Chi Tiết}
\label{sec:ch07-solutions}

\subsection{Lời Giải Nhóm A: Câu Hỏi Tư Duy Bản Chất}
\begin{itemize}
  \item \textbf{Lời giải Câu A.1}:
  Trong pha \textbf{Decode}, thao tác là Memory-bound với cường độ tính toán cực thấp ($1\text{ FLOP/Byte}$). GPU dành $99\%$ thời gian để nạp trọng số mô hình từ VRAM. Khi tăng Batch Size từ 1 lên 32, cùng một lần nạp trọng số mô hình từ VRAM có thể được tái sử dụng để tính toán cho 32 tokens khác nhau của 32 người dùng. Cường độ tính toán tăng gấp 32 lần, đưa hoạt động của GPU tiến gần hơn đến điểm bão hòa của mô hình Roofline, do đó GPU Utilization tăng vọt mà không tốn thêm bộ nhớ lưu trọng số.
  Ngược lại, trong pha \textbf{Prefill}, hệ thống thực hiện phép nhân ma trận $L \times L$ (Compute-bound). Kích thước bộ nhớ đệm Activation tỷ lệ thuận với $B \times L^2$. Nếu tăng Batch Size trong pha này khi prompt đã dài, ma trận kích hoạt trung gian vượt quá giới hạn SRAM/VRAM ngay lập tức, dẫn tới lỗi OOM.

  \item \textbf{Lời giải Câu A.2}:
  Trong các cuộc hội thoại thông thường, mỗi lượt chat chỉ kéo dài ngữ cảnh tuyến tính theo một nhánh duy nhất. Ngược lại, trong kỹ thuật Prompt Chaining và Tree-of-Thought (ToT), từ một nút suy luận ban đầu (Root Prompt), hệ thống rẽ nhánh sinh ra 5 đến 10 chuỗi suy luận khác nhau để tìm kiếm đường đi tối ưu. Nhờ cơ chế CoW của PagedAttention, tất cả các nhánh con đều dùng chung các block vật lý của câu hỏi gốc trong VRAM. Bộ nhớ tiết kiệm được tỷ lệ thuận với số lượng nhánh ($N-1$ lần dung lượng prompt gốc).

  \item \textbf{Lời giải Câu A.3}:
  Trong quá trình xây dựng từ vựng BPE trên dữ liệu internet, có những chuỗi ký tự kỳ lạ xuất hiện rất nhiều lần trong các tài liệu mã nguồn hoặc văn bản kỹ thuật (ví dụ: các ID người dùng Reddit, chuỗi byte rác lặp lại). Thuật toán BPE gộp chúng thành một token độc lập. Tuy nhiên, trong pha tiền huấn luyện mô hình (Pre-training), chuỗi ký tự này không bao giờ xuất hiện trong văn cảnh ngôn ngữ tự nhiên. Hệ quả là véc-tơ Embedding của token này không được học và nằm trôi dạt ở một vùng kỳ dị trong không gian đa chiều, khiến ma trận Attention sinh ra giá trị dị thường khi token này được kích hoạt.

  \item \textbf{Lời giải Câu A.4}:
  Speculative Decoding sử dụng thuật toán Modified Rejection Sampling: Với mỗi token $x$, xác suất chấp nhận được cân chỉnh chính xác bằng $\min\left(1, \frac{P_{\text{target}}(x)}{P_{\text{draft}}(x)}\right)$. Khi một token bị từ chối, phần bù xác suất được phân phối lại chính xác theo $\max(0, P_{\text{target}}(x) - P_{\text{draft}}(x))$. Theo định lý xác suất toàn phần, phân phối cận biên cuối cùng của token hoàn toàn trùng khớp với $P_{\text{target}}(x)$.

  \item \textbf{Lời giải Câu A.5}:
  Khi một prompt dài xuất hiện, nếu xử lý nguyên khối, nó sẽ độc chiếm Tensor Cores suốt hàng trăm mili-giây, chặn đứng mọi luồng Decode đang chạy. Chunked Prefill chia prompt thành các đoạn nhỏ (ví dụ 512 tokens), tại mỗi bước lặp chỉ xử lý 1 chunk và dành phần năng lực còn lại để phục vụ 1 token cho toàn bộ các luồng Decode. Nhờ đó, độ trễ ITL không bao giờ bị tăng đột biến.

  \item \textbf{Lời giải Câu A.6}:
  \textbf{PagedAttention} (vLLM) giải quyết bài toán phân mảnh bộ nhớ động trong phạm vi một request hoặc các nhánh song song của cùng một request (như Beam Search, Copy-on-Write). Khi request kết thúc, toàn bộ các trang KV Cache bị giải phóng, không duy trì trạng thái ngữ cảnh qua các request kế tiếp.
  Ngược lại, \textbf{RadixAttention} (SGLang) duy trì một cây tiền tố Radix Tree trên toàn bộ không gian VRAM, biến KV Cache thành kho tri thức sống giữa nhiều request độc lập. Khi request mới đến, giải thuật khớp tiền tố dài nhất (\code{match_prefix}) tái sử dụng $100\%$ các block KV Cache của phần tiền tố chung (System Prompt, Few-shot Examples, lịch sử Multi-turn Chat), đưa chi phí tính toán prefill của phần này về 0. Khi VRAM đầy, RadixAttention tự động thu hồi các nút lá ít truy cập nhất bằng LRU. Do đó, RadixAttention vượt trội trong các ứng dụng Agent lặp nhiều vòng, trích xuất thông tin cấu trúc ít shot, và các hệ thống chatbot doanh nghiệp có System Prompt đồ sộ.

  \item \textbf{Lời giải Câu A.7}:
  Kiến trúc \textbf{Medusa} (Cai et al., 2024) không cần mô hình Draft độc lập vì nó tích hợp các đầu dự đoán bổ sung (Medusa heads) trực tiếp trên tầng ẩn cuối cùng của mô hình Target. Mỗi đầu thứ $k$ học cách dự đoán token thứ $t+k$ một cách song song. Điều này loại bỏ hoàn toàn chi phí VRAM để nạp mô hình thứ hai và độ trễ truyền dữ liệu qua PCIe/NVLink.
  Để xác thực đồng thời nhiều chuỗi token phỏng đoán mà không tốn nhiều chu kỳ forward, Medusa tổ chức các ứng viên thành một cấu trúc cây (Candidate Tree) và áp dụng cơ chế \textbf{Tree Attention}. Ma trận mặt nạ Attention Mask được cấu hình để mỗi token chỉ chú ý đến các nút tổ tiên trên nhánh cây của nó. Nhờ đó, Target Model chỉ cần đúng \textbf{1 chu kỳ forward pass} để kiểm chứng toàn bộ các nhánh ứng viên song song.
\end{itemize}

\subsection{Lời Giải Nhóm B: Bài Toán Tính Toán Giải Tích Bằng Tay}

\paragraph{Lời giải Bài B.1}:
\begin{itemize}
  \item[a)] Áp dụng công thức tính dung lượng KV Cache cho Llama-3-70B dạng FP8 ($b=1$ byte):
  \begin{align*}
  M_{\text{token}} &= 2 \times n_{\text{layers}} \times n_{\text{kv\_heads}} \times d_{\text{head}} \times b \\
  &= 2 \times 80 \times 8 \times 128 \times 1 = 163.840\text{ Bytes} = 160\text{ KiB/token}
  \end{align*}
  Với chuỗi ngữ cảnh dài $L = 128.000$ tokens:
  \begin{equation*}
  M_{\text{session}} = 128.000 \times 163.840\text{ Bytes} = 20.971.520.000\text{ Bytes} \approx \mathbf{19.53\text{ GiB}}
  \end{equation*}

  \item[b)] Tổng VRAM của cụm 8 GPU H100 80GB: $V_{\text{total}} = 8 \times 80 = 640\text{ GB}$.
  Trừ đi trọng số mô hình FP8 ($70\text{ GB}$) và bộ nhớ đệm hệ thống ($20\text{ GB}$):
  \begin{equation*}
  V_{\text{available\_KV}} = 640 - 70 - 20 = 550\text{ GB}
  \end{equation*}
  Số lượng người dùng tối đa phục vụ đồng thời ở mức 128K context:
  \begin{equation*}
  B_{\max} = \left\lfloor \frac{550\text{ GB}}{19.53\text{ GB}} \right\rfloor = \mathbf{28\text{ người dùng đồng thời}}.
  \end{equation*}
\end{itemize}

\paragraph{Lời giải Bài B.2}:
Xét tỷ số xác suất giữa một token bất kỳ $i \ge 2$ và token lớn nhất $w_{(1)}$:
\begin{equation*}
\frac{P(w_{(i)})}{P(w_{(1)})} = \exp\left(\frac{z_{(i)} - z_{(1)}}{T}\right)
\end{equation*}
Vì $z_{(1)} > z_{(i)}$, đặt $\Delta_i = z_{(1)} - z_{(i)} > 0$. Khi $T \to 0^+$:
\begin{equation*}
\lim_{T \to 0^+} \exp\left(-\frac{\Delta_i}{T}\right) = \exp(-\infty) = 0
\end{equation*}
Do $\sum_{j=1}^n P(w_j) = 1 \implies P(w_{(1)}) \left[ 1 + \sum_{i=2}^n \frac{P(w_{(i)})}{P(w_{(1)})} \right] = 1$.
Lấy giới hạn $T \to 0^+$, ta có $\lim_{T \to 0^+} P(w_{(1)}) = 1$ và $\lim_{T \to 0^+} P(w_{(i)}) = 0$ với mọi $i \ge 2$. (Điều phải chứng minh).

\paragraph{Lời giải Bài B.3}:
Xác suất chấp nhận đúng $j$ token phỏng đoán ($0 \le j \le K-1$) là $\alpha^j (1 - \alpha)$. Khi đó hệ thống sinh được $j + 1$ tokens (gồm $j$ token draft và 1 token hiệu chỉnh).
Xác suất chấp nhận toàn bộ $K$ token phỏng đoán là $\alpha^K$. Khi đó hệ thống sinh được $K + 1$ tokens (gồm $K$ token draft và 1 token bổ sung).
Kỳ vọng số lượng token sinh ra sau 1 bước forward pass của Target Model:
\begin{align*}
\mathbb{E}[N] &= \sum_{j=0}^{K-1} (j + 1) \alpha^j (1 - \alpha) + (K + 1) \alpha^K \\
&= (1 - \alpha) \sum_{j=0}^{K-1} (j + 1) \alpha^j + (K + 1) \alpha^K = \frac{1 - \alpha^{K+1}}{1 - \alpha}
\end{align*}
Với $K = 4$ và $\alpha = 0.8$:
\begin{equation*}
\mathbb{E}[N] = \frac{1 - 0.8^5}{1 - 0.8} = \frac{1 - 0.32768}{0.2} = \frac{0.67232}{0.2} = \mathbf{3.3616\text{ tokens}}
\end{equation*}
Trong mỗi bước forward của Target Model, hệ thống thu được trung bình **3.36 tokens** thay vì chỉ 1 token! Tốc độ suy luận tăng hơn $3\times$ lần!

\paragraph{Lời giải Bài B.4}:
\begin{itemize}
  \item[a)] Xác suất Draft Model sinh ra $x$ và được Target Model chấp nhận trực tiếp là:
  \begin{equation*}
  P(\text{Draft } x \text{ được chấp nhận}) = q(x) \alpha(x) = q(x) \min\left(1, \frac{p(x)}{q(x)}\right) = \min(q(x), p(x))
  \end{equation*}
  Xác suất chấp nhận toàn phần tại pha Draft là tổng trên mọi $x \in \mathcal{V}$:
  \begin{equation*}
  \beta = \sum_{x \in \mathcal{V}} P(\text{Draft } x \text{ được chấp nhận}) = \sum_{x \in \mathcal{V}} \min(p(x), q(x))
  \end{equation*}

  \item[b)] Sử dụng hằng đẳng thức đại số: với mọi cặp số thực không âm $a, b$, ta có $\max(0, a - b) + \min(a, b) = a$.
  Áp dụng cho $a = p(y)$ và $b = q(y)$:
  \begin{equation*}
  \max(0, p(y) - q(y)) = p(y) - \min(p(y), q(y))
  \end{equation*}
  Lấy tổng trên toàn bộ không gian từ vựng $\mathcal{V}$:
  \begin{equation*}
  \sum_{y \in \mathcal{V}} \max(0, p(y) - q(y)) = \sum_{y \in \mathcal{V}} p(y) - \sum_{y \in \mathcal{V}} \min(p(y), q(y)) = 1 - \beta
  \end{equation*}

  \item[c)] Áp dụng công thức xác suất toàn phần:
  \begin{align*}
  P(X = x) &= P(\text{chấp nhận } x) + P(\text{từ chối}) \cdot p'(x) \\
  &= \min(p(x), q(x)) + (1 - \beta) \cdot \frac{\max(0, p(x) - q(x))}{1 - \beta} \\
  &= \min(p(x), q(x)) + \max(0, p(x) - q(x)) = p(x)
  \end{align*}
  Vậy phân phối xác suất cuối cùng $P(X = x) \equiv p(x)$ với mọi $x \in \mathcal{V}$. (Điều phải chứng minh).
\end{itemize}

\paragraph{Lời giải Bài B.5}:
\begin{itemize}
  \item[a)] Khi không có Cache tiền tố:
  Mỗi request cần tính toán lại toàn bộ $L = 2.000$ tokens:
  \begin{equation*}
  W_{\text{req}} = 2.000 \times 140\text{ GFLOPs} = 280.000\text{ GFLOPs} = 280\text{ TFLOPs}
  \end{equation*}
  Với tốc độ tiếp nhận $R = 100\text{ req/s}$, tổng công suất tính toán yêu cầu:
  \begin{equation*}
  W_{\text{total}} = 100 \times 280\text{ TFLOPs} = \mathbf{28.000\text{ TFLOPs/s}}
  \end{equation*}
  Hệ thống cần tối thiểu $\lceil 28.000 / 1.000 \rceil = 28$ GPU H100 riêng cho pha Prefill.
  Thời gian Prefill lý thuyết trên 1 GPU H100:
  \begin{equation*}
  \text{TTFT}_{\text{compute}} = \frac{280\text{ TFLOPs}}{1.000\text{ TFLOPs/s}} = 0.28\text{ s} = \mathbf{280\text{ ms}}
  \end{equation*}

  \item[b)] Khi Radix Tree đạt tỷ lệ khớp $85\%$ ($1.700$ tokens prefix được tái sử dụng):
  Số token cần tính toán prefill thực tế chỉ còn $L_{\text{new}} = 2.000 - 1.700 = 300\text{ tokens}$.
  Lượng tính toán cho mỗi request:
  \begin{equation*}
  W_{\text{req, cached}} = 300 \times 140\text{ GFLOPs} = 42\text{ TFLOPs}
  \end{equation*}
  Tổng công suất tính toán của cả cụm cho $100\text{ req/s}$:
  \begin{equation*}
  W_{\text{total, cached}} = 100 \times 42\text{ TFLOPs} = \mathbf{4.200\text{ TFLOPs/s}}
  \end{equation*}
  Số TFLOPs được giải phóng mỗi giây:
  \begin{equation*}
  \Delta W = 28.000 - 4.200 = \mathbf{23.800\text{ TFLOPs/s}}\quad (\text{tiết kiệm đúng } 85\%)
  \end{equation*}
  Thời gian Prefill trên 1 GPU H100 giảm xuống còn:
  \begin{equation*}
  \text{TTFT}_{\text{cached}} = \frac{42\text{ TFLOPs}}{1.000\text{ TFLOPs/s}} = 0.042\text{ s} = \mathbf{42\text{ ms}}
  \end{equation*}
  Độ trễ TTFT giảm tới \textbf{$6.67\times$ lần}, đồng thời hạ quy mô phần cứng từ 28 GPU xuống chỉ còn 5 GPU H100!
\end{itemize}

\subsection{Lời Giải Nhóm D: Phân Tích Thất Bại Thực Tế (Post-Mortem Analysis)}

\begin{debugbox}[Báo Cáo Sự Cố 1: Sập Cụm vLLM Do Prompt Khổng Lồ 100.000 Dòng]
\textbf{1. Tóm tắt sự cố}: Cụm phục vụ sập lúc 2:00 sáng với lỗi \code{CUDA OOM} do nhận một request duy nhất chứa file log 100.000 dòng ($>500\text{K}$ tokens).
\textbf{2. Phân tích nguyên nhân gốc rễ}:
- Thiếu tầng kiểm duyệt kích thước request tại API Gateway.
- Cấp phát bộ nhớ không giới hạn khiến một request đơn lẻ làm cạn kiệt toàn bộ Free Block Pool của PagedAttention.
\textbf{3. Giải pháp khắc phục}:
- Thiết lập giới hạn cứng tại Gateway: Trả về mã lỗi \code{HTTP 413} nếu prompt vượt quá $16.384$ tokens.
- Kích hoạt tham số \code{--max-model-len 32768} trong vLLM.
- Bật tính năng \code{--enable-chunked-prefill} để chia nhỏ prompt dài.
\end{debugbox}

\begin{debugbox}[Báo Cáo Sự Cố 2: Vỡ SLA Độ Trễ ITL Do Nghẽn Đầu Hàng (Head-of-Line Blocking)]
\textbf{1. Tóm tắt sự cố}: 50 người dùng đang chat mượt mà bị gián đoạn hơn 1.2 giây khi một người dùng khác gửi tài liệu 30.000 tokens.
\textbf{2. Phân tích nguyên nhân gốc rễ}:
Pha Prefill của tài liệu lớn chiếm dụng toàn bộ tài nguyên tính toán của GPU trong 1.2 giây, ngăn cản các bước Decode của 50 người dùng còn lại.
\textbf{3. Giải pháp khắc phục}:
- Triển khai \textbf{Chunked Prefill}: Chia tài liệu 30K thành các khối 512 tokens. Mỗi lượt tính toán, GPU xử lý 1 khối 512 tokens kèm 50 token decode của các khách hàng cũ. Độ trễ ITL duy trì ổn định dưới 30ms.
- Triển khai \textbf{Prefix Caching}: Nếu tài liệu hợp đồng được nhiều người dùng tra cứu lặp lại, vLLM tái sử dụng ngay các block KV Cache đã tính sẵn trong Radix Tree.
\end{debugbox}
